cmake_minimum_required(VERSION 3.16)
project(q3_amd_vulkan_rt_probe LANGUAGES CXX)
find_package(Vulkan 1.2 REQUIRED)
add_executable(q3-amd-vulkan-rt-probe main.cpp)
target_compile_features(q3-amd-vulkan-rt-probe PRIVATE cxx_std_11)
target_link_libraries(q3-amd-vulkan-rt-probe PRIVATE Vulkan::Vulkan)
if(MSVC)
  target_compile_options(q3-amd-vulkan-rt-probe PRIVATE /W4)
else()
  target_compile_options(q3-amd-vulkan-rt-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
